PostgreSQL Server Linux Commands:

    $ sudo systemctl start postgresql <-- (to start)

    $ sudo systemctl restart postgresql <-- (to re-start)
    
    $ sudo systemctl stop postgresql <-- (to Stop)
    
    $ sudo systemctl status postgresql <-- (to check status)

    $ sudo -u postgres psql <-- (to enter shell)

        $ CREATE USER username WITH PASSWORD; 'password' <-- (to create test_user) 

        $ ALTER USER username WITH CREATEDB; <-- (to privilege)

        $ ALTER USER username PASSWORD 'password'; <-- (to change password) 

        $ \du <-- (list all users)

        $ \q <-- (exit from sql)

        file = "src/schema.rs"

    main.rs
        use actix_web::{web, App, HttpServer, HttpResponse, Responder};
        use diesel::prelude::*;
        use diesel::r2d2::{self, ConnectionManager};

        // Define a Postgres connection pool
        type DbPool = r2d2::Pool<ConnectionManager<PgConnection>>;

        async fn index(pool: web::Data<DbPool>) -> impl Responder {
            // Acquire a connection from the pool
            let conn = pool.get().expect("couldn't get db connection from pool");

            // Example query: Fetch all records from a 'users' table
            let users = web::block(move || {
                use crate::schema::users::dsl::*;

                // Execute the query
                users.load::<User>(&conn)
            })
            .await
            .expect("Error loading users");

            // Format the result as a simple HTTP response
            HttpResponse::Ok().json(users)
        }

        #[actix_web::main]
        async fn main() -> std::io::Result<()> {
            dotenv::dotenv().ok();

            // Read the database URL from the environment variable
            let database_url = std::env::var("DATABASE_URL").expect("DATABASE_URL must be set");

            // Create a database connection manager for Diesel
            let manager = ConnectionManager::<PgConnection>::new(database_url);
            let pool = r2d2::Pool::builder()
                .build(manager)
                .expect("Failed to create pool");

            // Start Actix-Web server with a connection pool
            HttpServer::new(move || {
                App::new()
                    .data(pool.clone()) // Share the connection pool across handlers
                    .route("/", web::get().to(index))
            })
            .bind("127.0.0.1:8080")?
            .run()
            .await
        }
